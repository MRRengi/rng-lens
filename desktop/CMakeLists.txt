cmake_minimum_required(VERSION 3.16)
project(rng-lens VERSION 1.3.0 LANGUAGES C CXX)
set(RNGLENS_VERSION "" CACHE STRING "Version shown in the app (empty = the project version)")
if(NOT RNGLENS_VERSION)
    set(RNGLENS_VERSION "${PROJECT_VERSION}")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(RNGLENS_VCAM_ONLY "Build only the virtual camera DLL (used for the 32-bit DLL)" OFF)
set(TP ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
set(SOFTCAM ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/softcam)

# ---------------------------------------------------------------- virtual camera DLL (Windows)
if(WIN32)
  file(GLOB SOFTCAM_SRC ${SOFTCAM}/baseclasses/*.cpp ${SOFTCAM}/softcamcore/*.cpp)
  add_library(rnglens_vcam SHARED ${SOFTCAM_SRC} ${SOFTCAM}/softcam/softcam.cpp ${SOFTCAM}/softcam/softcam.def)
  target_include_directories(rnglens_vcam PRIVATE ${SOFTCAM} ${SOFTCAM}/baseclasses)
  target_compile_definitions(rnglens_vcam PRIVATE UNICODE _UNICODE NDEBUG)
  target_link_libraries(rnglens_vcam PRIVATE strmiids ole32 oleaut32 uuid winmm user32 gdi32 advapi32)
  set_target_properties(rnglens_vcam PROPERTIES PREFIX "")
  if(MINGW)
    target_compile_options(rnglens_vcam PRIVATE -include ${SOFTCAM}/mingw_compat.h -fpermissive -w)
    target_link_options(rnglens_vcam PRIVATE -static -static-libgcc -static-libstdc++ -s)
  endif()
  # The optional second phone's camera, "rng-lens 2": same code, own CLSID and shared memory.
  add_library(rnglens_vcam2 SHARED ${SOFTCAM_SRC} ${SOFTCAM}/softcam/softcam.cpp ${SOFTCAM}/softcam/softcam.def)
  target_include_directories(rnglens_vcam2 PRIVATE ${SOFTCAM} ${SOFTCAM}/baseclasses)
  target_compile_definitions(rnglens_vcam2 PRIVATE UNICODE _UNICODE NDEBUG RNGLENS_VCAM_SECOND)
  target_link_libraries(rnglens_vcam2 PRIVATE strmiids ole32 oleaut32 uuid winmm user32 gdi32 advapi32)
  set_target_properties(rnglens_vcam2 PROPERTIES PREFIX "")
  if(MINGW)
    target_compile_options(rnglens_vcam2 PRIVATE -include ${SOFTCAM}/mingw_compat.h -fpermissive -w)
    target_link_options(rnglens_vcam2 PRIVATE -static -static-libgcc -static-libstdc++ -s)
  endif()
endif()
if(RNGLENS_VCAM_ONLY)
  return()
endif()

# ---------------------------------------------------------------- dependencies
include(FetchContent)
if(NOT WIN32)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale libswresample)
  pkg_check_modules(GLFW IMPORTED_TARGET glfw3)
endif()
if(WIN32 OR NOT GLFW_FOUND)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  if(EXISTS ${GLFW_SOURCE_DIR})
    add_subdirectory(${GLFW_SOURCE_DIR} glfw)
  else()
    FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz)
    FetchContent_MakeAvailable(glfw)
  endif()
  set(GLFW_TARGET glfw)
else()
  set(GLFW_TARGET PkgConfig::GLFW)
endif()

if(WIN32)
  if(NOT FFMPEG_ROOT)
    set(FFMPEG_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/ffmpeg-win)
  endif()
  add_library(ffmpeg INTERFACE)
  target_include_directories(ffmpeg INTERFACE ${FFMPEG_ROOT}/include)
  target_link_libraries(ffmpeg INTERFACE ${FFMPEG_ROOT}/lib/libavcodec.dll.a ${FFMPEG_ROOT}/lib/libavutil.dll.a
                        ${FFMPEG_ROOT}/lib/libswscale.dll.a ${FFMPEG_ROOT}/lib/libswresample.dll.a)
  set(FFMPEG_TARGET ffmpeg)
else()
  set(FFMPEG_TARGET PkgConfig::FFMPEG)
endif()

# ---------------------------------------------------------------- app
set(IMGUI ${TP}/imgui)
file(GLOB FONT_FILES ${CMAKE_CURRENT_SOURCE_DIR}/res/fonts/*.ttf)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/fonts_embed.cpp
  COMMAND ${CMAKE_COMMAND} -DSRC=${CMAKE_CURRENT_SOURCE_DIR}/res/fonts -DDST=${CMAKE_CURRENT_BINARY_DIR}/fonts_embed.cpp
          -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_fonts.cmake
  DEPENDS ${FONT_FILES} ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_fonts.cmake)
add_executable(rng-lens
  src/main.cpp src/app.cpp src/headless.cpp src/connection.cpp src/decoder.cpp src/discovery.cpp
  src/platform.cpp src/prefs.cpp src/renderer.cpp src/audio.cpp src/miniaudio_impl.c src/vision.cpp src/shell.cpp src/stb_impl.cpp src/vcam_windows.cpp src/vcam_linux.cpp
  src/ui.cpp src/updater.cpp ${CMAKE_CURRENT_BINARY_DIR}/fonts_embed.cpp
  ${TP}/glad/src/gl.c
  ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
  ${IMGUI}/backends/imgui_impl_glfw.cpp ${IMGUI}/backends/imgui_impl_opengl3.cpp)
target_include_directories(rng-lens PRIVATE src ${TP} ${TP}/glad/include ${IMGUI})
target_compile_definitions(rng-lens PRIVATE RNGLENS_VERSION="${RNGLENS_VERSION}")
target_link_libraries(rng-lens PRIVATE ${FFMPEG_TARGET} ${GLFW_TARGET})
if(WIN32)
  target_sources(rng-lens PRIVATE res/rng-lens.rc)
  target_link_libraries(rng-lens PRIVATE ws2_32 shell32 ole32 comdlg32 dwmapi advapi32 winhttp)
  if(MINGW)
    target_link_options(rng-lens PRIVATE -mwindows -static -static-libgcc -static-libstdc++ -s)
  endif()
else()
  find_package(Threads REQUIRED)
  target_link_libraries(rng-lens PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
